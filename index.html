/* ========================================================= 港鐵巴士 – 更穩健版本（解決 CORS / 找不到路線） ========================================================= */ var mtrStopsCache = null; // { routeId: [ {id, name, dir, seq} ] } var mtrRouteNamesCache = null; var mtrScheduleCache = {}; // route -> last schedule response function fetchWithCorsFallback(url, options) { options = options || {}; return fetch(url, options).then(function(res) { if (!res.ok) throw new Error('HTTP ' + res.status); return res; }).catch(function() { // CORS 失敗時用 allorigins 代理 var proxyUrl = 'https://api.allorigins.win/raw?url=' + encodeURIComponent(url); if (options.method === 'POST' && options.body) { // POST 無法簡單走 raw proxy，改用 get 包裝（部分環境可用） // 實際改用 corsproxy 或直接再試一次 return fetch('https://corsproxy.io/?' + encodeURIComponent(url), options); } return fetch(proxyUrl).then(function(res) { if (!res.ok) throw new Error('proxy fail'); return res; }); }); } function getMtrStopsData(callback) { if (mtrStopsCache) { callback(mtrStopsCache, mtrRouteNamesCache || {}); return; } var csvUrl = 'https://opendata.mtr.com.hk/data/mtr_bus_stops.csv'; fetchWithCorsFallback(csvUrl) .then(function(res) { return res.text(); }) .then(function(csvText) { mtrStopsCache = {}; var lines = csvText.split(/\r?\n/); for (var i = 1; i < lines.length; i++) { var line = lines[i].trim(); if (!line) continue; var parts = line.split(','); if (parts.length < 8) continue; var routeId = parts[0].replace(/^\uFEFF/, '').trim(); var direction = parts[1].trim(); var seq = parseInt(parts[2], 10) || 0; var stationId = parts[3].trim(); var nameChi = parts[6].trim(); if (!mtrStopsCache[routeId]) mtrStopsCache[routeId] = []; mtrStopsCache[routeId].push({ id: stationId, name: nameChi, dir: direction, seq: seq }); } callback(mtrStopsCache, mtrRouteNamesCache || {}); }) .catch(function() { mtrStopsCache = mtrStopsCache || {}; callback(mtrStopsCache, {}); }); } function getMTRBusSchedule(routeName, callback) { var route = String(routeName || '').toUpperCase().trim(); if (!route) { callback(null); return; } if (mtrScheduleCache[route] && (Date.now() - mtrScheduleCache[route]._ts < 20000)) { callback(mtrScheduleCache[route]); return; } var body = JSON.stringify({ language: 'zh', routeName: route }); fetch('https://rt.data.gov.hk/v1/transport/mtr/bus/getSchedule', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body }) .then(function(res) { if (!res.ok) throw new Error('status ' + res.status); return res.json(); }) .then(function(data) { data._ts = Date.now(); mtrScheduleCache[route] = data; callback(data); }) .catch(function() { // CORS / 網路失敗 → 用代理 var proxyUrl = 'https://corsproxy.io/?' + encodeURIComponent('https://rt.data.gov.hk/v1/transport/mtr/bus/getSchedule'); fetch(proxyUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body }) .then(function(res) { return res.json(); }) .then(function(data) { data._ts = Date.now(); mtrScheduleCache[route] = data; callback(data); }) .catch(function() { callback(null); }); }); } function getMTRBusETA(routeName, stopId, callback) { if (!routeName || !stopId) { callback([]); return; } getMTRBusSchedule(routeName, function(data) { var list = []; if (data && data.busStop && Array.isArray(data.busStop)) { for (var i = 0; i < data.busStop.length; i++) { var stopObj = data.busStop[i]; if (String(stopObj.busStopId) !== String(stopId)) continue; if (stopObj.isSuspended === '1') continue; var buses = stopObj.bus || []; for (var j = 0; j < buses.length; j++) { var b = buses[j]; var sec = parseInt(b.arrivalTimeInSecond, 10); if (isNaN(sec) || sec < 0) { if (sec === 0 || (b.arrivalTimeText && /即將|Arriving|Departed|將到/.test(b.arrivalTimeText))) { list.push({ eta: new Date().toISOString(), arrivalText: '將到站', mins: 0 }); } continue; } var etaDate = new Date(Date.now() + sec * 1000); list.push({ eta: etaDate.toISOString(), arrivalText: b.arrivalTimeText || '', mins: Math.round(sec / 60) }); } break; } } list.sort(function(a, b) { return new Date(a.eta) - new Date(b.eta); }); callback(list); }); }
